<script>

function hidePages()
{
    document.getElementById("home").style.display = "none";
    document.getElementById("events").style.display = "none";
    document.getElementById("booking").style.display = "none";
    document.getElementById("payment").style.display = "none";
    document.getElementById("success").style.display = "none";
    document.getElementById("mybooking").style.display = "none";
}


function showHome()
{
    hidePages();

    document.getElementById("home").style.display = "block";
}


function showEvents()
{
    hidePages();

    document.getElementById("events").style.display = "block";
}


function showBooking()
{
    hidePages();

    document.getElementById("booking").style.display = "block";
}


function showPayment()
{
    hidePages();

    document.getElementById("payment").style.display = "block";
}


function showMyBooking()
{
    hidePages();

    document.getElementById("mybooking").style.display = "block";
}


function selectMusic()
{
    document.getElementById("bookingEvent").innerHTML = "Music Fest";

    document.getElementById("bookingDate").innerHTML =
    "📅 30 Sept 2026";

    document.getElementById("bookingLocation").innerHTML =
    "📍 Pune, Maharashtra";

    document.getElementById("bookingPrice").innerHTML =
    "499";

    showBooking();
}


function selectLive()
{
    document.getElementById("bookingEvent").innerHTML = "Live Show";

    document.getElementById("bookingDate").innerHTML =
    "📅 05 Oct 2026";

    document.getElementById("bookingLocation").innerHTML =
    "📍 Pune, Maharashtra";

    document.getElementById("bookingPrice").innerHTML =
    "699";

    showBooking();
}


function selectComedy()
{
    document.getElementById("bookingEvent").innerHTML =
    "Comedy Night";

    document.getElementById("bookingDate").innerHTML =
    "📅 10 Oct 2026";

    document.getElementById("bookingLocation").innerHTML =
    "📍 Pune, Maharashtra";

    document.getElementById("bookingPrice").innerHTML =
    "399";

    showBooking();
}


function confirmBooking()
{
    var name =
    document.getElementById("name").value;

    var mobile =
    document.getElementById("mobile").value;

    var tickets =
    document.getElementById("tickets").value;


    if(name == "" || mobile == "")
    {
        alert("Please enter all details");
    }

    else
    {
        var event =
        document.getElementById("bookingEvent").innerHTML;

        var date =
        document.getElementById("bookingDate").innerHTML;

        var price =
        document.getElementById("bookingPrice").innerHTML;

        var location =
        document.getElementById("bookingLocation").innerHTML;


        var total = price * tickets;


        /* PAYMENT DETAILS */

        document.getElementById("paymentName").innerHTML =
        name;

        document.getElementById("paymentEvent").innerHTML =
        event;

        document.getElementById("paymentDate").innerHTML =
        date;

        document.getElementById("paymentTickets").innerHTML =
        tickets;

        document.getElementById("paymentPrice").innerHTML =
        price;

        document.getElementById("paymentTotal").innerHTML =
        total;


        /* MY BOOKING DETAILS */

        document.getElementById("myName").innerHTML =
        name;

        document.getElementById("myMobile").innerHTML =
        mobile;

        document.getElementById("myEvent").innerHTML =
        event;

        document.getElementById("myDate").innerHTML =
        date;

        document.getElementById("myLocation").innerHTML =
        location;

        document.getElementById("myTickets").innerHTML =
        tickets;

        document.getElementById("myTotal").innerHTML =
        total;


        /* EVENT IMAGE */

        if(event == "Music Fest")
        {
            document.getElementById("myImage").src =
            "https://images.unsplash.com/photo-1492684223066-81342ee5ff30?auto=format&fit=crop&w=600&q=80";
        }

        else if(event == "Live Show")
        {
            document.getElementById("myImage").src =
            "https://images.unsplash.com/photo-1501386761578-eac5c94b800a?auto=format&fit=crop&w=600&q=80";
        }

        else
        {
            document.getElementById("myImage").src =
            "https://images.unsplash.com/photo-1585699324551-f6c309eedeca?auto=format&fit=crop&w=600&q=80";
        }


        showPayment();
    }
}


function makePayment()
{
    alert("Payment Successful!");

    hidePages();

    document.getElementById("success").style.display =
    "block";
}
